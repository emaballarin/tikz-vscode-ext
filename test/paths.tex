% Path syntax whose words are also English. Each operator is used the way TikZ
% uses it, and the same words appear in node text, where they are only text.
\begin{tikzpicture}
  \draw (a) .. controls (b) and (c) .. (d);
  \draw (a) ..controls (b)..(d);
  \draw (a) .. controls +(1,0) and +(-1,0) .. (d);
  \foreach \x in {1,...,5} \fill (\x,0) circle (1pt);
  \draw (a) to[bend left] (b) to (c);
  \draw (0,0) circle (1cm) (2,0) circle [radius=5mm];
  \draw (0,0) grid (2,2);
  \draw (0,0) rectangle (1,1) -- cycle;
  \draw (0,0) arc (0:90:1cm);
  \path (0,0) node {plain} coordinate (x);
  \draw (a) edge[->] (b);
  \foreach \x in {1,2} \draw (\x,0) -- +(1,0) -- ++(0,1);
  \draw (0,0) ellipse (2cm and 1cm) (3,0) parabola bend (4,1) (5,0);
  \draw plot coordinates {(0,0) (1,1)} plot[smooth] file {data.table};
  \path let \p1 = (a) in (\p1) circle (1pt);
  \node {root} child {node {leaf}} child foreach \x in {1,2} {node {\x}};
  \draw decorate [decoration=zigzag] {(0,0) -- (1,0)} (2,0) svg {M 0 0 L 1 1};
  \path pic (p) at (0,0) {angle} graph {a -> b} foreach \x in {1,2} {(\x,0) circle (1pt)};
  \node {plot a graph, decorate the svg, let it in, pick a pic, then cycle};
  \node {go to the store in town at noon};
  \node {a circle and a grid in a rectangle};
\end{tikzpicture}
